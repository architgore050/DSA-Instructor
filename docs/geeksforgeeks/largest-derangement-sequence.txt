Title: Largest Derangement of a Sequence
Extractor: 2

Given any sequence S, find the largest derangement . A derangement D is any permutation such that no two elements at the same position in S and D are equal.  The "largest derangement" of a sequence refers to the derangement that results in the lexicographically largest permutation

Examples:

> Input : seq[] = {5, 4, 3, 2, 1} Output : 4 5 2 1 3 Input : seq[] = {56, 21, 42, 67, 23, 74} Output : 74, 67, 56, 42, 21, 23

Since we are interested in generating the largest derangement, we start putting larger elements in more significant positions. Start from left, at any position i place the next largest element among the values of the sequence which have not yet been placed in positions before.

To scan all positions takes n iteration. In each iteration we are required to find a maximum number, so a trivial implementation would be O(n^2) complexity, However, if we use a data structure like max-heap to find the maximum element, then the complexity reduces to O(n Log n)

C++

#include <bits/stdc++.h>
using namespace std;

void printLargest(vector<int> seq)
{
    int n = seq.size();
    vector<int> res(n); // Stores result

    // Insert all elements into a priority queue
    priority_queue<int> pq;
    for (int i = 0; i < n; i++)
        pq.push(seq[i]);

    // Fill Up res[] from left to right
    for (int i = 0; i < n; i++) {
        int d = pq.top();
        pq.pop();
        if (d != seq[i] || i == n - 1) {
            res[i] = d;
        } else {

            // New Element popped equals the element
            // in original sequence. Get the next
            // largest element
            res[i] = pq.top();
            pq.pop();
            pq.push(d);
        }
    }

    // If given sequence is in descending order then
    // we need to swap last two elements again
    if (res[n - 1] == seq[n - 1]) {
        res[n - 1] = res[n - 2];
        res[n - 2] = seq[n - 1];
    }

    printf("\nLargest Derangement \n");
    for (int i = 0; i < n; i++)
        printf("%d ", res[i]);
}

// Driver code
int main()
{
    vector<int> seq = { 92, 3, 52, 13, 2, 31, 1 };
    printLargest(seq);
    return 0;
}

from queue import PriorityQueue

def print_largest(seq):
    n = len(seq)
    res = [0] * n  # Stores result

    # Insert all elements into a priority queue
    pq = PriorityQueue()
    for i in range(n):
        pq.put(-seq[i])  # Using negative
                         # values to simulate max-heap

    # Fill up res[] from left to right
    for i in range(n):
        d = -pq.get()
        if d != seq[i] or i == n - 1:
            res[i] = d
        else:
            # New Element popped equals the element
            # in original sequence. Get the next
            # largest element
            res[i] = -pq.get()
            pq.put(-d)

    # If given sequence is in descending order then
    # we need to swap last two elements again
    if res[n - 1] == seq[n - 1]:
        res[n - 1] = res[n - 2]
        res[n - 2] = seq[n - 1]

    print("\nLargest Derangement")
    for i in range(n):
        print(res[i], end=' ')

C#

using System;
using System.Collections.Generic;

public class MainClass {
    static void PrintLargest(int[] seq) {
        int n = seq.Length;
        int[] res = new int[n]; // Stores result
        Array.Copy(seq, res, n);

        // Insert all elements into a priority queue
        SortedSet<int> pq = new SortedSet<int>(seq);

        // Fill Up res[] from left to right
        for (int i = 0; i < n; i++) {
            int d = pq.Min;
            pq.Remove(d);
            if (d != seq[i] || i == n - 1) {
                res[i] = d;
            } else {
                // New Element popped equals the element
                // in original sequence. Get the next
                // largest element
                int nextLargest = pq.GetViewBetween(d + 1, int.MaxValue).Min;
                res[i] = nextLargest;
                pq.Remove(nextLargest);
                pq.Add(d);
            }
        }

        // If given sequence is in descending order then
        // we need to swap last two elements again
        if (res[n - 1] == seq[n - 1]) {
            res[n - 1] = res[n - 2];
            res[n - 2] = seq[n - 1];
        }

        Console.WriteLine("\nLargest Derangement ");
        for (int i = 0; i < n; i++)
            Console.Write(res[i] + " ");
    }

    // Driver code
    public static void Main(string[] args) {
        int[] seq = {92, 3, 52, 13, 2, 31, 1};
        PrintLargest(seq);
    }
}

JavaScript

const { MaxPriorityQueue } = require('@datastructures-js/priority-queue');

function printLargest(seq) {
    const n = seq.length;
    let res = new Array(n);

    // Create max priority queue and enqueue all elements
    let pq = new MaxPriorityQueue();
    for (let i = 0; i < n; i++) {
        pq.enqueue(seq[i]);
    }

    // Build result array
    for (let i = 0; i < n; i++) {
        let d = pq.front().element;
        pq.dequeue();

        if (d !== seq[i] || i === n - 1) {
            res[i] = d;
        } else {
            // If top of PQ is same as original, use next top
            res[i] = pq.front().element;
            pq.dequeue();
            pq.enqueue(d);  // Push back the original top
        }
    }

    // Handle descending case
    if (res[n - 1] === seq[n - 1]) {
        [res[n - 2], res[n - 1]] = [res[n - 1], res[n - 2]];
    }

    // Output result
    console.log("\nLargest Derangement\n");
    console.log(res.join(" "));
}

// Example usage
let seq = [92, 3, 52, 13, 2, 31, 1];
printLargest(seq);

Output

Largest Derangement
52 92 31 3 13 1 2

Time Complexity: O(n log n) Auxiliary Space: O(n), because, we use an N size array to store results.

Note: The method can be easily modified to obtain the smallest derangement as well.  Instead of a Max Heap , we should use a Min Heap to consecutively get minimum elements
