Title: Count numbers with exactly 3 divisors
Extractor: 2

You are given an integer array queries[] , where each element represents a query value n. For every n in the array, determine how many numbers less than or equal to n have exactly 3 divisors.

Examples:

> Input: queries[] = [6] Output: [1] Explanation: There is only one number 4 which has exactly three divisors 1, 2 and 4 and less than equal to 6. Input: query[] = [3, 10] Output: [0, 2] Explanation: For query[0], the answer is 0. For query[1], there are two numbers 4 and 9 having exactly 3 divisors.

Table of Content

- [Naive Approach] Divisor Count Check - O(q × n × √n) Time and O(1) Space
- [Better Approach] Prime Square Property - O(q × √n × √√n) Time and O(1) Space
- [Expected Approach] Sieve with Prefix Count - O(√M log log √M + q) Time and O(√M) Space

### [Naive Approach] Divisor Count Check - O(q × n × √n) Time and O(1) Space

> For each query, iterate all numbers from 1 to n and count divisors of each number. If divisor count equals 3, increment total.

- For each query, initialize total = 0
- For num from 1 to n, compute divisor count
- If divisor count equals 3, increment total
- Store total in answer array
- Return answer array

C++

#include <iostream>
#include <vector>
using namespace std;

// Count total divisors of a number
int divisorCount(int num) {
    int cnt = 0;

    for (int i = 1; i <= num; i++) {
        if (num % i == 0) {
            cnt++;
        }
    }

    return cnt;
}

// For each query, count numbers having exactly 3 divisors
vector<int> threeDivisors(vector<int>& queries) {
    vector<int> ans;

    for (int n : queries) {

        int total = 0;

        // Check every number from 1 to n
        for (int num = 1; num <= n; num++) {

            if (divisorCount(num) == 3) {
                total++;
            }
        }

        ans.push_back(total);
    }

    return ans;
}

int main() {
    vector<int> queries = {6, 10, 100};

    vector<int> ans = threeDivisors(queries);

    for (int x : ans) {
        cout << x << " ";
    }

    return 0;
}

### [Better Approach] Prime Square Property - O(q × √n × √√n) Time and O(1) Space

> A number has exactly 3 divisors only if it is square of a prime. So count primes p such that p² ≤ n .

Why the number having exactly 3 divisors must be a perfect square ?

For a number to have exactly 3 divisors, the divisors must be:

- 1 (every number is divisible by 1)
- The number itself (a number is always divisible by itself)
- One other divisor between 1 and the number.

All divisors appear in pairs. Only in case of squares, we can have n/d = d.

Why it must be a Perfect Square of Prime?

If the third divisor is not a prime, then there would be more three divisors. Hence it must be a prime.

                total++;

### [Expected Approach] Sieve with Prefix Count - O(√M log log √M + q) Time and O(√M) Space

> A number has exactly 3 divisors only if it is square of a prime. Use Sieve to mark primes up to √M, build prefix count, and answer each query in O(1).

- Find maximum query value M
- Compute limit = √M and run Sieve up to limit
- Build prefixPrime array where prefixPrime[i] = number of primes ≤ i
- For each query n, compute root = √n and return prefixPrime[root]

C++

#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

vector<int> threeDivisors(vector<int>& queries) {

    // Find maximum query value
    int maxQuery = 0;

    for (int x : queries) {
        maxQuery = max(maxQuery, x);
    }

    // We only need primes up to sqrt(maxQuery)
    int limit = sqrt(maxQuery);

    // Sieve of Eratosthenes
    vector<bool> isPrime(limit + 1, true);
    isPrime[0] = isPrime[1] = false;

    for (int i = 2; i * i <= limit; i++) {

        if (isPrime[i]) {

            // Mark multiples as non-prime
            for (int j = i * i; j <= limit; j += i) {
                isPrime[j] = false;
            }
        }
    }

    // prefixPrime[i] = number of primes <= i
    vector<int> prefixPrime(limit + 1, 0);

    for (int i = 1; i <= limit; i++) {
        prefixPrime[i] = prefixPrime[i - 1] + (isPrime[i] ? 1 : 0);
    }

    vector<int> ans;

    for (int n : queries) {

        // Numbers having exactly 3 divisors
        // = prime squares <= n
        int root = sqrt(n);

        ans.push_back(prefixPrime[root]);
    }

    return ans;
}

    vector<int> queries = {6, 10, 100};

    vector<int> ans = threeDivisors(queries);

    for (int x : ans) {
        cout << x << " ";
    }

    return 0;
}

Java

import java.util.*;

class Solution {

    static List<Integer> threeDivisors(List<Integer> query) {

        // Find the largest query value
        int maxQuery = 0;
        for (int n : query) {
            maxQuery = Math.max(maxQuery, n);
        }

        // We only need primes up to sqrt(maxQuery)
        int limit = (int) Math.sqrt(maxQuery);

        // Sieve of Eratosthenes
        boolean[] isPrime = new boolean[limit + 1];
        Arrays.fill(isPrime, true);

        if (limit >= 0) isPrime[0] = false;
        if (limit >= 1) isPrime[1] = false;

        // Mark all non-prime numbers
        for (int i = 2; i * i <= limit; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= limit; j += i) {
                    isPrime[j] = false;
                }
            }
        }

        // prefixPrime[i] stores count of primes <= i
        int[] prefixPrime = new int[limit + 1];

        for (int i = 1; i <= limit; i++) {
            prefixPrime[i] = prefixPrime[i - 1];
            if (isPrime[i]) {
                prefixPrime[i]++;
            }
        }

        List<Integer> ans = new ArrayList<>();

        // For each query:
        // Count prime numbers <= sqrt(n)
        // Every such prime contributes one prime square having exactly 3 divisors
        for (int n : query) {
            int root = (int) Math.sqrt(n);
            ans.add(prefixPrime[root]);
        }

        return ans;
    }

    public static void main(String[] args) {

        List<Integer> query = Arrays.asList(6, 10, 100);

        List<Integer> ans = threeDivisors(query);

        for (int x : ans) {
            System.out.print(x + " ");
        }
    }
}

# Python program to count numbers having exactly 3 divisors using Sieve
import math

def threeDivisors(queries):
    # Find maximum query value
    maxQuery = max(queries)

    # We only need primes up to sqrt(maxQuery)
    limit = int(math.sqrt(maxQuery))

    # Sieve of Eratosthenes
    isPrime = [True] * (limit + 1)
    if limit >= 0:
        isPrime[0] = False
    if limit >= 1:
        isPrime[1] = False

    i = 2
    while i * i <= limit:
        if isPrime[i]:
            # Mark multiples as non-prime
            for j in range(i * i, limit + 1, i):
                isPrime[j] = False
        i += 1

    # prefixPrime[i] = number of primes <= i
    prefixPrime = [0] * (limit + 1)
    for i in range(1, limit + 1):
        prefixPrime[i] = prefixPrime[i - 1] + (1 if isPrime[i] else 0)

    ans = []

    for n in queries:
        # Numbers having exactly 3 divisors = prime squares <= n
        root = int(math.sqrt(n))
        ans.append(prefixPrime[root])

    return ans

    static List<int> threeDivisors(int[] queries) {

        foreach (int x in queries) {
            maxQuery = Math.Max(maxQuery, x);
        }

        // We only need primes up to sqrt(maxQuery)
        int limit = (int)Math.Sqrt(maxQuery);

        // Sieve of Eratosthenes
        bool[] isPrime = new bool[limit + 1];
        for (int i = 0; i <= limit; i++) {
            isPrime[i] = true;
        }
        if (limit >= 0) isPrime[0] = false;
        if (limit >= 1) isPrime[1] = false;

        for (int i = 2; i * i <= limit; i++) {
            if (isPrime[i]) {
                // Mark multiples as non-prime
                for (int j = i * i; j <= limit; j += i) {
                    isPrime[j] = false;
                }
            }

        for (int i = 1; i <= limit; i++) {
            prefixPrime[i] = prefixPrime[i - 1] + (isPrime[i] ? 1 : 0);

        List<int> ans = new List<int>();

        foreach (int n in queries) {
