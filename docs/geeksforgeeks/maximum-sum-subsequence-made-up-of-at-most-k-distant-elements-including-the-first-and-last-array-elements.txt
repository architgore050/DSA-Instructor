Title: Maximum sum subsequence made up of at most K distant elements including the first and last array elements
Extractor: 2

Given an array arr[] of size n , and an integer k . The task is to find the maximum sum of a subsequence that satisfies the following two conditions :

1. The subsequence must include both arr[0] and arr[n - 1] .
2. The indices of consecutive elements chosen in the subsequence must differ by at most k .

Examples:

> Input: arr[] = [10, -5, -2, 4, 0, 3], k = 3 Output: 17 Explanation: Start at arr[0] = 10 → jump to arr[3] = 4 → jump to arr[5] = 3. Total sum = 10 + 4 + 3 = 17. Input: arr[] = [1, -5, -20, 4, -1, 3, -6, -3], k = 2 Output: 0 Explanation: Best path starting from arr[0] = 1 to arr[7] = -3 using max jump k = 2 gives maximum possible sum = 0. Input: arr[] = [-1, -2, -3, -4], k = 1 Output: -10

Table of Content

- [Naive Approach] Generate and Check All Subsequences - O(k^(n/k)) Time and O(n) Space
- [Expected Approach] Using DP + Deque - O(n) Time and O(n) Space

### [Naive Approach] Generate and Check All Subsequences - O(k^(n/k)) Time and O(n) Space

> The idea is to explore all possible valid subsequences starting from arr[0] and ending at arr[n - 1], where each selected element's index is within k steps from the previous one. The thought process is to use recursion to try every path, accumulate the sum, and update the maximum sum found. At each recursive call, we try jumping to every index from idx + 1 to idx + k if it’s within bounds. The base case is when we reach the last index, at which point we check if the current sum is higher than the current maximum. Backtracking helps undo the current choice and try alternate paths, ensuring complete exploration of possibilities.

C++

// C++ program to find the maximum sum subsequence
// with adjacent elements have at most k index
// distance using Naive Approach
#include <bits/stdc++.h>
using namespace std;

// Recursive function to explore all valid subsequences
void explore(int idx, vector<int>& arr, int k, int sum,
             vector<int>& path, int& maxSum) {

    // If the last element is reached
    if (idx == arr.size() - 1) {
        maxSum = max(maxSum, sum);
        return;
    }

    // Try all next possible positions within k distance
    for (int next = idx + 1;
         next <= idx + k && next < arr.size(); next++) {

        // Include arr[next] in the subsequence
        path.push_back(arr[next]);

        // Recur to explore further from next index
        explore(next, arr, k, sum + arr[next], path, maxSum);

        // Backtrack to try other possibilities
        path.pop_back();
    }
}

// Function to initiate the recursion and return max sum
int maxSumSubsequence(vector<int>& arr, int k) {

    int n = arr.size();

    if (n < 2) {
        return -1;
    }

    int maxSum = INT_MIN;

    // Start the path from arr[0]
    vector<int> path;
    path.push_back(arr[0]);

    explore(0, arr, k, arr[0], path, maxSum);

    return maxSum;
}

int main() {

    vector<int> arr = {10, -5, -2, 4, 0, 3};
    int k = 3;

    cout << maxSumSubsequence(arr, k);

    return 0;
}

### [Expected Approach] Using DP + Deque - O(n) Time and O(n) Space

> The idea is to use Dynamic Programming with a Deque to efficiently maintain the maximum sum reachable at each index. The thought process is: for every index i , we want the maximum dp[j] (where i-k ≤ j < i) to add arr[i] to it. We use a Deque to store potential j indices in a way that always keeps the maximum at the front, allowing O(1) access. The key observation is that we can discard smaller dp values from the back of the deque since they’ll never contribute to a better solution.

Steps-By-Step Implementation

- Create a dp array to store the maximum sum that can reach each position while satisfying the constraints.
- Initialize the first position in dp with the value of the first element from the input array.
- Use a deque to maintain indices of elements with useful maximum values in the last k steps.
- For each index from 1 to n - 1, remove indices from the deque that are out of the valid k-range .
- Compute the current dp value as the maximum of the window (deque front) plus the current element.
- Remove indices from the back of the deque that have dp values less than or equal to the current one.
- After processing all elements, return the value at the last position in the dp array as the answer.

C++

// C++ program to find the maximum sum subsequence
// with adjacent elements have at most k index
// distance using Expected Approach (Deque + DP)
#include <bits/stdc++.h>
using namespace std;

int maxSumSubsequence(vector<int>& arr, int k) {

    int n = arr.size();

    if (n < 2) {
        return -1;
    }

    // dp[i] stores the max sum to reach index i
    vector<int> dp(n, INT_MIN);
    dp[0] = arr[0];

    // Deque to store indices of max dp values in window
    deque<int> dq;
    dq.push_back(0);

    for (int i = 1; i < n; i++) {

        // Remove indices out of window [i - k, i - 1]
        while (!dq.empty() && dq.front() < i - k) {
            dq.pop_front();
        }

        // Current dp[i] is max dp in window + arr[i]
        dp[i] = dp[dq.front()] + arr[i];

        // Maintain decreasing order in deque
        while (!dq.empty() && dp[i] >= dp[dq.back()]) {
            dq.pop_back();
        }

        // Add current index
        dq.push_back(i);
    }

    return dp[n - 1];
}

int main() {

    vector<int> arr = {10, -5, -2, 4, 0, 3};
    int k = 3;

    cout << maxSumSubsequence(arr, k);

    return 0;
}

from collections import deque

def maxSumSubsequence(arr, k):

    n = len(arr)

    if n < 2:
        return -1

    # dp[i] stores the max sum to reach index i
    dp = [float('-inf')] * n
    dp[0] = arr[0]

    # Deque to store indices of max dp values in window
    dq = deque()
    dq.append(0)

    for i in range(1, n):

        # Remove indices out of window [i - k, i - 1]
        while dq and dq[0] < i - k:
            dq.popleft()

        # Current dp[i] is max dp in window + arr[i]
        dp[i] = dp[dq[0]] + arr[i]

        # Maintain decreasing order in deque
        while dq and dp[i] >= dp[dq[-1]]:
            dq.pop()

        # Add current index
        dq.append(i)

    return dp[n - 1]

// C# program to find the maximum sum subsequence
// with adjacent elements have at most k index
// distance using Expected Approach (Deque + DP)
using System;
using System.Collections.Generic;

class GfG {

    public static int maxSumSubsequence(int[] arr, int k) {

        int n = arr.Length;

        if (n < 2) {
            return -1;
        }

        // dp[i] stores the max sum to reach index i
        int[] dp = new int[n];
        for (int i = 0; i < n; i++) dp[i] = int.MinValue;
        dp[0] = arr[0];

        // Deque to store indices of max dp values in window
        LinkedList<int> dq = new LinkedList<int>();
        dq.AddLast(0);

        for (int i = 1; i < n; i++) {

            // Remove indices out of window [i - k, i - 1]
            while (dq.Count > 0 && dq.First.Value < i - k) {
                dq.RemoveFirst();
            }

            // Current dp[i] is max dp in window + arr[i]
            dp[i] = dp[dq.First.Value] + arr[i];

            // Maintain decreasing order in deque
            while (dq.Count > 0 && dp[i] >= dp[dq.Last.Value]) {
                dq.RemoveLast();
            }

            // Add current index
            dq.AddLast(i);
        }

        return dp[n - 1];
    }

    public static void Main() {

        int[] arr = {10, -5, -2, 4, 0, 3};
        int k = 3;

        Console.WriteLine(maxSumSubsequence(arr, k));
    }
}

    return dp[n - 1];
